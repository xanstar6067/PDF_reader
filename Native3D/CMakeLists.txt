cmake_minimum_required(VERSION 3.20)
project(PaperDesk3D LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(NANO "${CMAKE_CURRENT_SOURCE_DIR}/../third-party/nanoprc/source")
set(ZLIB "${CMAKE_CURRENT_SOURCE_DIR}/../third-party/zlib/source")
file(GLOB ZLIB_SRC "${ZLIB}/*.c")
list(FILTER ZLIB_SRC EXCLUDE REGEX "/(example|minigzip)\\.c$")
add_library(zlib STATIC ${ZLIB_SRC})
target_include_directories(zlib PUBLIC "${ZLIB}")
file(GLOB PRC_SRC "${NANO}/src/*.c")
add_library(prc STATIC ${PRC_SRC})
file(WRITE "${CMAKE_BINARY_DIR}/prc_version.h" "#define PRC_VERSION \"PaperDesk-v1\"\n#define PRC_GIT_DESCRIBE \"vendored-source\"\n")
target_include_directories(prc PUBLIC "${NANO}/include" "${CMAKE_BINARY_DIR}")
target_compile_definitions(prc PRIVATE PRC_LITTLE_ENDIAN PRC_VERSION_STRING="PaperDesk-v1" PRC_DEBUG_MEMORY=0 PRC_ENABLE_DIAG_ENV=0 PRC_ENABLE_UNZIPPED_FUZZ=0)
target_link_libraries(prc PRIVATE zlib)
add_executable(PaperDeskModelWorker "../third-party/nanoprc/paperdesk-worker.cpp")
target_link_libraries(PaperDeskModelWorker PRIVATE prc zlib)
add_executable(PaperDeskPrcTests EXCLUDE_FROM_ALL "../third-party/nanoprc/paperdesk-prc-tests.c")
target_link_libraries(PaperDeskPrcTests PRIVATE prc zlib)
add_library(PaperDesk3D SHARED Renderer.cpp)
target_compile_definitions(PaperDesk3D PRIVATE UNICODE _UNICODE NOMINMAX)
target_link_libraries(PaperDesk3D PRIVATE d3d11 dxgi d3dcompiler)
